Install

Drop it in.
Flip it on.

CocoaPods 
pod 'DoSwiftKit',
  :git => 'https://github.com/darkThanBlack/DoSwiftKit.git',
  :branch => 'main',
  :tag => '1.0.0'
SwiftPM 
.package(
  url: "https://github.com/darkThanBlack/DoSwiftKit.git",
  from: "1.0.0"
)
Initialization 
// SceneDelegate.swift
#if DEBUG
DO.Core.shared.setup(appScene: windowScene, appWindow: window)
DO.NetworkPlugin.shared.start()
DO.ConsolePlugin.shared.start()
#endif

// AppDelegate.swift (without scenes)
#if DEBUG
DO.Core.shared.setup(appScene: window?.windowScene, appWindow: window)
#endif
